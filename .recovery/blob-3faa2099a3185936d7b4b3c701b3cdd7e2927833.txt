    [Fact]
    public async Task ThreeSequentialChangesAreAppendOnlyAndUseCurrentRevisionAsBaseline()
    {
        var seed = await SeedOrder(); var original = seed.Snapshot;
        await CompleteChange(seed, seed.ExpensiveVariant); await CompleteChange(seed, seed.MidVariant); await CompleteChange(seed, seed.ExpensiveVariant);
        await using var db = fixture.CreateDbContext(); var item = await db.Set<OrderItem>().SingleAsync(x => x.Id == seed.ItemId); var revisions = await db.Set<OrderItemRevision>().Where(x => x.OrderItemId == seed.ItemId).OrderBy(x => x.RevisionNumber).ToListAsync();
        Assert.Equal(3, item.CurrentRevisionNumber); Assert.Equal(original, item.Snapshot); Assert.Equal([1, 2, 3], revisions.Select(x => x.RevisionNumber)); Assert.Equal([(12.50m, 20m, 7.50m), (20m, 15m, -5m), (15m, 20m, 5m)], revisions.Select(x => (x.PreviousUnitAmount, x.UnitAmount, x.PriceDifference)));
        var revisionPayloads = revisions.Select(x => x.Snapshot).ToArray(); var productId = await db.Set<Appizza.Modules.Catalog.ProductVariant>().Where(x => x.Id == seed.CheapVariant).Select(x => x.ProductId).SingleAsync(); var product = await db.Set<Appizza.Modules.Catalog.Product>().SingleAsync(x => x.Id == productId); product.Name = "Nome posterior"; product.Description = "Descrição posterior"; product.Status = "archived"; product.Version++; var variants = await db.Set<Appizza.Modules.Catalog.ProductVariant>().Where(x => x.ProductId == productId).ToListAsync(); foreach (var variant in variants) { variant.BasePrice += 100; variant.Version++; } await db.SaveChangesAsync(); db.ChangeTracker.Clear();
        Assert.Equal(original, await db.Set<OrderItem>().Where(x => x.Id == seed.ItemId).Select(x => x.Snapshot).SingleAsync()); Assert.Equal(revisionPayloads, await db.Set<OrderItemRevision>().Where(x => x.OrderItemId == seed.ItemId).OrderBy(x => x.RevisionNumber).Select(x => x.Snapshot).ToArrayAsync());
    }

    private async Task CompleteChange(Seed seed, Guid variant)
    {
        var response = await Change(seed, variant, Guid.NewGuid()); response.EnsureSuccessStatusCode(); using var json = JsonDocument.Parse(await response.Content.ReadAsStringAsync()); if (json.RootElement.GetProperty("status").GetString() != "pending_customer_confirmation") return; var confirmed = await fixture.PostWithIdempotencyAsync($"api/v1/table-device/order-item-requests/{json.RootElement.GetProperty("requestId").GetGuid()}/confirm-review", new { requestVersion = json.RootElement.GetProperty("requestVersion").GetInt32(), reviewHash = json.RootElement.GetProperty("reviewHash").GetString(), accepted = true }, seed.DeviceToken, Guid.NewGuid()); confirmed.EnsureSuccessStatusCode();
    }

    private Task<HttpResponseMessage> Change(Seed seed, Guid variant, Guid key) => fixture.PostWithIdempotencyAsync($"api/v1/table-device/order-items/{seed.ItemId}/change-requests", new { configuration = new { productVariantId = variant }, reasonCode = "CUSTOMER_CHANGE", expectedOrderItemVersion = (long?)null }, seed.DeviceToken, key);

    private async Task EnsureProduction(Seed seed, string status)
    {
        await using var db = fixture.CreateDbContext(); var stationId = await db.Set<Appizza.Modules.Kitchen.Station>().Where(x => x.EstablishmentId == seed.TenantId && x.IsDefault).Select(x => x.Id).FirstAsync(); var now = DateTimeOffset.UtcNow; db.Add(new Appizza.Modules.Kitchen.ProductionItem { Id = Guid.NewGuid(), EstablishmentId = seed.TenantId, OrderItemId = seed.ItemId, StationId = stationId, Status = status, ReceivedAt = now, CreatedAt = now, UpdatedAt = now }); await db.SaveChangesAsync();
    }
